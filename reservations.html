วิธีที่ 1 — Apps Script (แนะนำ)

เชื่อมต่อโดยตรงกับ Sheet แบบ Private ดึงข้อมูลสด ทุกครั้งที่กด Sync

📄 Apps Script Code — วางใน Extensions → Apps Script ของ Sheet 
function doGet(e) {
  var ss = SpreadsheetApp.openById('1XbTJLhecql_HNqyE80Hc6h30A2_elIxliudF4e6Rlz0');
  var sheet = ss.getSheetByName('Sheet1');
  var data = sheet.getDataRange().getValues();

  var bookings = [];
  var ledger = [];
  var summary = {};
  var mode = 'header';

  for (var i = 0; i < data.length; i++) {
    var row = data[i];
    var c0 = String(row[0] || '').trim();

    if (c0 === 'เลขห้อง') { mode = 'bookings'; continue; }
    if (c0.indexOf('สรุปยอด') === 0 || c0 === 'Booking.com') { mode = 'summary'; }
    if (c0 === 'วันที่ตรวจพบ') { mode = 'ledger'; continue; }

    if (mode === 'bookings' && c0 && c0.indexOf('สรุป') !== 0) {
      bookings.push({
        room: c0,
        guest: String(row[1] || ''),
        checkin: String(row[2] || ''),
        checkout: String(row[3] || ''),
        channel: String(row[4] || ''),
        resId: String(row[5] || ''),
        note: String(row[6] || '')
      });
    }

    if (mode === 'summary' && c0 && row[1]) {
      summary[c0] = parseFloat(String(row[1]).replace(/,/g, '')) || 0;
    }

    if (mode === 'ledger' && c0 && c0 !== 'วันที่ตรวจพบ') {
      ledger.push({
        date: String(row[0] || ''),
        ota: String(row[1] || ''),
        bookingId: String(row[2] || ''),
        guest: String(row[4] || ''),
        room: String(row[5] || ''),
        checkin: String(row[6] || ''),
        checkout: String(row[7] || ''),
        nights: parseInt(row[8]) || 0,
        gross: parseFloat(String(row[9]).replace(/,/g, '')) || 0,
        commission: parseFloat(String(row[10]).replace(/,/g, '')) || 0,
        net: parseFloat(String(row[11]).replace(/,/g, '')) || 0,
        status: String(row[12] || '')
      });
    }
  }

  var result = { bookings: bookings, ledger: ledger, summary: summary };
  return ContentService
    .createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}
วิธี Deploy:
1. เปิด Google Sheet → Extensions → Apps Script
2. วางโค้ดด้านบน → Save
3. Deploy → New Deployment → Web App
4. Execute as: Me   Who has access: Anyone
5. Copy URL → วางในช่องด้านบน → กด เชื่อมต่อ
